    <script>

        const MAX_RPM = 180;

        

        // Lo stato del sistema

        let state = {

            posX: 0, posY: 0,

            angle: 0, 

            currRpm: 0, targetRpm: 0,

            turn: 0,

            v: 4.2, a: 0.05

        };


        const world = document.getElementById('world-plane');

        const chassis = document.getElementById('chassis');

        const needle = document.getElementById('needle');

        const speedoFill = document.getElementById('speedo-fill');


        function update() {

            // Fisica Fluida

            state.currRpm += (state.targetRpm - state.currRpm) * 0.08;

            const speedPerc = Math.abs(state.currRpm) / MAX_RPM;

            

            // Angolazione e Direzione

            state.angle += state.turn * (speedPerc * 4 || 1.5);

            

            // Calcolo scorrimento (Infinite scroll logic)

            const rad = (state.angle - 90) * (Math.PI / 180);

            const moveStep = (state.currRpm / MAX_RPM) * 15;

            

            state.posX -= Math.cos(rad) * moveStep;

            state.posY -= Math.sin(rad) * moveStep;


            // Applichiamo lo scorrimento e la rotazione al pavimento

            // Nota: Il rover resta fisso al centro, ruotiamo il mondo per simulare la curva

            world.style.backgroundPosition = `${state.posX}px ${state.posY}px`;

            world.style.transform = `rotateX(70deg) rotateZ(${-state.angle}deg)`;


            // Animazioni Rover (Beccheggio e Rollio)

            const pitch = (state.targetRpm - state.currRpm) * 0.2;

            const roll = state.turn * 3;

            chassis.style.transform = `rotateX(${pitch}deg) rotateY(${roll}deg)`;


            // Tachimetro (Arco 210 gradi)

            const arc = speedPerc * 148;

            speedoFill.style.strokeDasharray = `${arc} 360`;

            needle.style.transform = `rotate(${-130 + (speedPerc * 210)}deg)`;

            document.getElementById('speed-text').innerText = (speedPerc * 1.85).toFixed(2);


            // Telemetria

            state.a += ((0.05 + speedPerc * 1.5) - state.a) * 0.1;

            state.v -= state.a * 0.000002;

            document.getElementById('volt-text').innerText = state.v.toFixed(2);

            document.getElementById('amp-text').innerText = state.a.toFixed(2);

            document.getElementById('batt-level').style.width = `${((state.v-3.2)/(4.2-3.2)*100)}%`;


            requestAnimationFrame(update);

        }


        function cmd(type) {

            switch(type) {

                case 'AVANTI': state.targetRpm = MAX_RPM; state.turn = 0; break;

                case 'INDIETRO': state.targetRpm = -MAX_RPM; state.turn = 0; break;

                case 'DESTRA': state.turn = 3; state.targetRpm = MAX_RPM * 0.7; break;

                case 'SINISTRA': state.turn = -3; state.targetRpm = MAX_RPM * 0.7; break;

                case 'STOP': state.targetRpm = 0; state.turn = 0; break;

                case 'LIGHTS':

                    const beams = document.querySelectorAll('.beam');

                    const isOff = beams[0].style.display === 'none' || beams[0].style.display === '';

                    beams.forEach(b => b.style.display = isOff ? 'block' : 'none');

                    break;

                

            }

        }


        const setup = (id, c) => {

            const el = document.getElementById(id);

            const start = (e) => { e.preventDefault(); cmd(c); el.classList.add('active'); };

            const end = () => { el.classList.remove('active'); if(c !== 'STOP') cmd('STOP'); };

            el.onmousedown = start; el.ontouchstart = start;

            el.onmouseup = end; el.ontouchend = end; el.onmouseleave = end;

        };


        setup('btn-up','AVANTI'); setup('btn-down','INDIETRO');

        setup('btn-left','SINISTRA'); setup('btn-right','DESTRA');

        document.getElementById('btn-stop').onclick = () => cmd('STOP');

        document.getElementById('btn-lights').onclick = () => cmd('LIGHTS');


        requestAnimationFrame(update);

    </script>
